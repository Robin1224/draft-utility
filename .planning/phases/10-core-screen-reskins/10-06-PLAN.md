    Rewrite PauseOverlay.svelte. Remove ALL Tailwind and the `import TimerDisplay`. KEEP the props `let { captainName, graceEndsAt, timerMs = 30000 } = $props();`.

    Add a self-contained grace countdown (mirror the proven TimerDisplay tick, but inline so there is no cross-file coupling):
    ```js
    let secondsLeft = $state(0);
    $effect(() => {
      function tick() {
        secondsLeft = Math.max(0, Math.ceil((new Date(graceEndsAt).getTime() - Date.now()) / 1000));
      }
      tick();
      const id = setInterval(tick, 250);
      return () => clearInterval(id);
    });
    const totalGraceSeconds = $derived(Math.round(timerMs / 1000));
    const barWidth = $derived(timerMs > 0 ? Math.max(0, Math.min(100, (secondsLeft * 1000 / timerMs) * 100)) : 100);
    ```
